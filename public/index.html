const express = require('express'); const cors = require('cors'); const morgan = require('morgan'); const path = require('path'); const dotenv = require('dotenv'); dotenv.config(); const app = express(); const PORT = process.env.PORT || 5000; app.use(cors()); app.use(express.json()); app.use(express.urlencoded({ extended: true })); app.use(morgan('dev')); const productRoutes = require('./routes/products'); const userRoutes = require('./routes/users'); const orderRoutes = require('./routes/orders'); app.use('/api/products', productRoutes); app.use('/api/users', userRoutes); app.use('/api/orders', orderRoutes); app.get('/api/health', (req, res) => { res.json({ success: true, message: 'Amazon store backend is running', timestamp: new Date().toISOString() }); }); const publicDir = path.join(__dirname, '../public'); app.use(express.static(publicDir)); app.get('*', (req, res) => { res.sendFile(path.join(publicDir, 'index.html')); }); app.listen(PORT, () => { console.log(`Server running on http://localhost:${PORT}`); });
